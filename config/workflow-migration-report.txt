name: Fix Workflow Config Paths

on:
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: workflow-path-migration
  cancel-in-progress: false

jobs:
  fix-workflow-config-paths:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Scan and fix all workflow paths
        run: |
          node <<'EOF'
          const fs = require('fs')
          const path = require('path')

          const workflowsDir = '.github/workflows'
          const reportPath = 'config/workflow-path-migration-report.txt'

          fs.mkdirSync('config', { recursive: true })

          const replacements = [
            {
              oldPath: 'guides/.tvmaze-fast-cache.json',
              newPath: 'config/.tvmaze-fast-cache.json',
              type: 'exact'
            },
            {
              oldPath: 'guides/24-7-channel-list.txt',
              newPath: 'config/24-7-channel-list.txt',
              type: 'exact'
            },
            {
              oldPath: 'guides/iptvorg-channels.txt',
              newPath: 'config/iptvorg-channels.txt',
              type: 'exact'
            },
            {
              oldPath: 'fastchannels.txt',
              newPath: 'config/fastchannels.txt',
              type: 'bare'
            },
            {
              oldPath: 'channels.txt',
              newPath: 'config/channels.txt',
              type: 'bare'
            },
            {
              oldPath: 'iptvorg-channels.txt',
              newPath: 'config/iptvorg-channels.txt',
              type: 'bare'
            },
            {
              oldPath: 'generate_247.py',
              newPath: 'config/generate_247.py',
              type: 'bare'
            },
            {
              oldPath: 'generate_matched_guide.py',
              newPath: 'config/generate_matched_guide.py',
              type: 'bare'
            },
            {
              oldPath: 'generate_sports.py',
              newPath: 'config/generate_sports.py',
              type: 'bare'
            },
            {
              oldPath: 'matcheverychannel.txt',
              newPath: 'config/matcheverychannel.txt',
              type: 'bare'
            },
            {
              oldPath: 'metadata_cache.json',
              newPath: 'config/metadata_cache.json',
              type: 'bare'
            },
            {
              oldPath: 'sports_channels.txt',
              newPath: 'config/sports_channels.txt',
              type: 'bare'
            }
          ]

          const report = []

          let workflowFiles = []
          let totalFilesScanned = 0
          let totalFilesChanged = 0
          let totalReplacements = 0
          let totalNotFound = 0
          let totalAlreadyCorrect = 0

          const replacementResults = new Map()

          for (const item of replacements) {
            replacementResults.set(item.oldPath, {
              oldPath: item.oldPath,
              newPath: item.newPath,
              occurrences: 0,
              files: [],
              found: false,
              alreadyCorrect: false
            })
          }

          function getWorkflowFiles(dir) {
            if (!fs.existsSync(dir)) return []

            return fs.readdirSync(dir)
              .filter(file => /\.(yml|yaml)$/i.test(file))
              .map(file => path.join(dir, file))
          }

          function escapeRegExp(value) {
            return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
          }

          function countOccurrences(text, regex) {
            const matches = text.match(regex)
            return matches ? matches.length : 0
          }

          workflowFiles = getWorkflowFiles(workflowsDir)
          totalFilesScanned = workflowFiles.length

          report.push('============================================================')
          report.push('WORKFLOW CONFIG PATH MIGRATION REPORT')
          report.push('============================================================')
          report.push('')
          report.push(`Generated: ${new Date().toISOString()}`)
          report.push('')
          report.push(`Workflow files scanned: ${totalFilesScanned}`)
          report.push('')

          if (workflowFiles.length === 0) {
            report.push('[ERROR] No workflow files were found.')
            report.push('')
            fs.writeFileSync(reportPath, report.join('\n') + '\n')
            console.error(report.join('\n'))
            process.exit(1)
          }

          for (const workflowFile of workflowFiles) {
            let content = fs.readFileSync(workflowFile, 'utf8')
            const originalContent = content
            const fileChanges = []

            for (const item of replacements) {
              const result = replacementResults.get(item.oldPath)

              const exactRegex = new RegExp(
                escapeRegExp(item.oldPath),
                'g'
              )

              const alreadyCorrectRegex = new RegExp(
                escapeRegExp(item.newPath),
                'g'
              )

              const alreadyCorrectCount = countOccurrences(
                content,
                alreadyCorrectRegex
              )

              if (alreadyCorrectCount > 0) {
                result.alreadyCorrect = true
                totalAlreadyCorrect += alreadyCorrectCount
              }

              let oldRegex

              if (item.type === 'exact') {
                oldRegex = exactRegex
              } else {
                oldRegex = new RegExp(
                  `(?<![A-Za-z0-9_./-])${escapeRegExp(item.oldPath)}(?![A-Za-z0-9_./-])`,
                  'g'
                )
              }

              const matches = content.match(oldRegex)
              const count = matches ? matches.length : 0

              if (count > 0) {
                content = content.replace(oldRegex, item.newPath)

                result.found = true
                result.occurrences += count
                result.files.push(workflowFile)

                totalReplacements += count

                fileChanges.push({
                  oldPath: item.oldPath,
                  newPath: item.newPath,
                  count
                })
              }
            }

            if (content !== originalContent) {
              fs.writeFileSync(workflowFile, content)
              totalFilesChanged++

              report.push(`[CHANGED] ${workflowFile}`)
              report.push('')

              for (const change of fileChanges) {
                report.push(`  [REPLACED ${change.count} TIME(S)]`)
                report.push(`    OLD: ${change.oldPath}`)
                report.push(`    NEW: ${change.newPath}`)
                report.push('')
              }
            } else {
              report.push(`[UNCHANGED] ${workflowFile}`)
              report.push('  No old paths required replacement.')
              report.push('')
            }
          }

          report.push('============================================================')
          report.push('REPLACEMENTS MADE')
          report.push('============================================================')
          report.push('')

          let foundAny = false

          for (const item of replacements) {
            const result = replacementResults.get(item.oldPath)

            if (!result.found) continue

            foundAny = true

            report.push(`[REPLACED] ${result.oldPath}`)
            report.push(`    → ${result.newPath}`)
            report.push(`    Occurrences: ${result.occurrences}`)
            report.push('    Workflow files:')

            for (const file of [...new Set(result.files)]) {
              report.push(`      - ${file}`)
            }

            report.push('')
          }

          if (!foundAny) {
            report.push('No old paths were found and replaced.')
            report.push('')
          }

          report.push('============================================================')
          report.push('NOT FOUND / MISSED OLD PATHS')
          report.push('============================================================')
          report.push('')

          for (const item of replacements) {
            const result = replacementResults.get(item.oldPath)

            if (!result.found) {
              totalNotFound++

              report.push(`[NOT FOUND] ${result.oldPath}`)
              report.push(`    Expected replacement: ${result.newPath}`)
              report.push('')
            }
          }

          report.push('============================================================')
          report.push('ALREADY CORRECT / SKIPPED')
          report.push('============================================================')
          report.push('')

          let alreadyCorrectAny = false

          for (const item of replacements) {
            const result = replacementResults.get(item.oldPath)

            if (!result.alreadyCorrect) continue

            alreadyCorrectAny = true

            report.push(`[ALREADY CORRECT] ${result.newPath}`)
            report.push('    Existing config path was detected.')
            report.push('')
          }

          if (!alreadyCorrectAny) {
            report.push('No already-correct paths were detected.')
            report.push('')
          }

          report.push('============================================================')
          report.push('SUMMARY')
          report.push('============================================================')
          report.push('')
          report.push(`Workflow files scanned: ${totalFilesScanned}`)
          report.push(`Workflow files changed: ${totalFilesChanged}`)
          report.push(`Workflow files unchanged: ${totalFilesScanned - totalFilesChanged}`)
          report.push(`Total replacements made: ${totalReplacements}`)
          report.push(`Old paths not found: ${totalNotFound}`)
          report.push(`Already-correct path detections: ${totalAlreadyCorrect}`)
          report.push('Errors: 0')
          report.push('')
          report.push('============================================================')

          fs.writeFileSync(reportPath, report.join('\n') + '\n')

          console.log(report.join('\n'))
          console.log('')
          console.log(`Report written to: ${reportPath}`)
          EOF

      - name: Show migration report
        run: |
          echo "============================================================"
          echo "FINAL WORKFLOW PATH MIGRATION REPORT"
          echo "============================================================"
          cat config/workflow-path-migration-report.txt

      - name: Commit and push workflow changes and report
        run: |
          git config user.name "github-actions"
          git config user.email "github-actions@github.com"

          git add .github/workflows/*.yml
          git add .github/workflows/*.yaml
          git add config/workflow-path-migration-report.txt

          if git diff --cached --quiet; then
            echo "No workflow path changes or report changes to commit"
            exit 0
          fi

          git commit -m "Update workflow paths for config files"

          for attempt in 1 2 3 4 5; do
            echo "Push attempt $attempt"

            if git push origin HEAD:main; then
              echo "Push succeeded"
              exit 0
            fi

            echo "Push failed, fetching latest main and rebasing..."

            git fetch origin main
            git rebase origin/main

            git add .github/workflows/*.yml
            git add .github/workflows/*.yaml
            git add config/workflow-path-migration-report.txt

            if git diff --cached --quiet; then
              echo "No new changes after rebase"
            else
              git commit -m "Update workflow paths for config files"
            fi

            sleep 5
          done

          echo "Push failed after all retries"
          exit 1
